import sys
import json
import os

import pandas as pd
import joblib


# ==================================================
# PATHS
# ==================================================

BASE_DIR = os.path.dirname(
    os.path.dirname(
        os.path.abspath(__file__)
    )
)

CROP_MODEL_PATH = os.path.join(
    BASE_DIR,
    "models",
    "crop_recommendation_model.pkl"
)

COMPATIBILITY_MODEL_PATH = os.path.join(
    BASE_DIR,
    "models",
    "soil_compatibility_model.pkl"
)

SOIL_CLIMATE_PATH = os.path.join(
    BASE_DIR,
    "data",
    "soil_climate.csv"
)


# ==================================================
# LOAD MODELS
# ==================================================

crop_model_data = joblib.load(
    CROP_MODEL_PATH
)

compatibility_model_data = joblib.load(
    COMPATIBILITY_MODEL_PATH
)

crop_model = crop_model_data["model"]

compatibility_model = (
    compatibility_model_data["model"]
)


# ==================================================
# CROP NAME MAPPING
# ==================================================

CROP_MAPPING = {
    "rice": "Rice",
    "maize": "Maize",
    "pigeonpeas": "Arhar",
    "mungbean": "Moong",
    "blackgram": "Urd",
    "lentil": "Masoor",
    "chickpea": "Gram"
}


def map_crop_name(crop):

    return CROP_MAPPING.get(
        crop.lower(),
        crop
    )


# ==================================================
# LOAD DATASET
# ==================================================

soil_climate_df = pd.read_csv(
    SOIL_CLIMATE_PATH
)

available_crops = set(
    soil_climate_df[
        "Crop_Type"
    ]
    .astype(str)
    .str.lower()
)


# ==================================================
# MAIN RECOMMENDATION FUNCTION
# ==================================================

def recommend(data):

    # ------------------------------------------------
    # 1. CROP MODEL INPUT
    # ------------------------------------------------

    crop_input = pd.DataFrame([{

        "N": data["nitrogen"],

        "P": data["phosphorus"],

        "K": data["potassium"],

        "temperature":
            data["temperature"],

        "humidity":
            data["humidity"],

        "ph":
            data["ph"],

        "rainfall":
            data["rainfall"]

    }])


    # ------------------------------------------------
    # 2. GET CROP PROBABILITIES
    # ------------------------------------------------

    probabilities = (
        crop_model
        .predict_proba(
            crop_input
        )[0]
    )

    classes = crop_model.classes_


    ranked = sorted(
        zip(
            classes,
            probabilities
        ),
        key=lambda x: x[1],
        reverse=True
    )


    # ------------------------------------------------
    # 3. KEEP ONLY REAL CANDIDATES
    # ------------------------------------------------

    candidates = []

    for crop, probability in ranked:

        confidence = (
            float(probability) * 100
        )

        # Ignore extremely low predictions
        if confidence < 1:
            continue

        mapped_crop = map_crop_name(
            crop
        )

        # Only use crops available in
        # compatibility dataset
        if (
            mapped_crop.lower()
            not in available_crops
        ):
            continue

        candidates.append({

            "crop": crop,

            "dataset_crop":
                mapped_crop,

            "ml_confidence":
                round(
                    confidence,
                    2
                )
        })

        # Maximum 5 serious candidates
        if len(candidates) >= 5:
            break


    # ------------------------------------------------
    # 4. COMPATIBILITY CHECK
    # ------------------------------------------------

    recommendations = []


    for candidate in candidates:

        compatibility_input = pd.DataFrame([{

            "Crop_Type":
                candidate["dataset_crop"],

            "Soil_Type":
                data["soil_type"],

            "Farm_Size_Acres":
                data["farm_size_acres"],

            "Irrigation_Available":
                data[
                    "irrigation_available"
                ],

            "Soil_pH":
                data["ph"],

            "Soil_Nitrogen":
                data["nitrogen"],

            "Soil_Organic_Matter":
                data[
                    "soil_organic_matter"
                ],

            "Temperature":
                data["temperature"],

            "Rainfall":
                data["rainfall"],

            "Humidity":
                data["humidity"]

        }])


        compatibility_probabilities = (
            compatibility_model
            .predict_proba(
                compatibility_input
            )[0]
        )


        compatibility_classes = (
            compatibility_model
            .classes_
        )


        probability_map = {

            int(cls):
                float(prob)

            for cls, prob in zip(
                compatibility_classes,
                compatibility_probabilities
            )

        }


        compatible_probability = (
            probability_map.get(
                1,
                0
            ) * 100
        )


        not_compatible_probability = (
            probability_map.get(
                0,
                0
            ) * 100
        )


        # ------------------------------------------------
        # 5. FINAL SCORE
        # ------------------------------------------------

        ml_score = (
            candidate[
                "ml_confidence"
            ]
        )

        compatibility_score = (
            compatible_probability
        )


        final_score = (
            ml_score * 0.75
            +
            compatibility_score * 0.25
        )


        recommendations.append({

            "crop":
                candidate["crop"],

            "dataset_crop":
                candidate[
                    "dataset_crop"
                ],

            "ml_confidence":
                candidate[
                    "ml_confidence"
                ],

            "compatibility_confidence":
                round(
                    compatible_probability,
                    2
                ),

            "not_compatible_confidence":
                round(
                    not_compatible_probability,
                    2
                ),

            "final_score":
                round(
                    final_score,
                    2
                ),

            "compatible":
                compatible_probability >= 50

        })


    # ------------------------------------------------
    # 6. SORT RESULTS
    # ------------------------------------------------

    recommendations.sort(

        key=lambda x:
            x["final_score"],

        reverse=True

    )


    # ------------------------------------------------
    # 7. RETURN
    # ------------------------------------------------

    return {

        "input":
            data,

        "recommendations":
            recommendations

    }


# ==================================================
# PROGRAM ENTRY
# ==================================================

if __name__ == "__main__":

    try:

        input_json = sys.stdin.read()

        if not input_json.strip():

            raise ValueError(
                "No JSON input received"
            )

        data = json.loads(
            input_json
        )

        result = recommend(
            data
        )

        print(
            json.dumps(
                result,
                indent=2
            )
        )

    except Exception as error:

        print(
            json.dumps({
                "error":
                    str(error)
            })
        )
